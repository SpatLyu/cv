%% ---------------------------------------------------------------------------
%% vitae: entry macros
%%
%% Upstream sets its listings as bare `tabular`s built by an R helper. These
%% macros take the same two-column shape -- a date at the left, the entry
%% beside it -- but as paragraphs rather than a table, so that a long section
%% breaks across pages between entries. The `entries/*.tex` listings in this
%% extension are what call them.
%%
%% A `tabular` sizes a column to its widest cell for free, which is why
%% upstream never had a column width to set. Paragraphs do not, so each
%% listing measures its own labels first: `entries/*.tex` loops over the
%% entries twice, once calling \cvmeasure and once emitting them. That keeps
%% the column tight to the dates a section actually has -- narrow for years,
%% wider for spans -- with nothing to configure.
%% ---------------------------------------------------------------------------

% Gap between the label column and the entry beside it.
\newlength{\cvlabelsep}
\setlength{\cvlabelsep}{0.75em}

\newlength{\cvcolwidth}   % widest label seen so far in this listing
\newlength{\cvcoltotal}   % that, plus the gap: the column the entry starts at
\newlength{\cvlabelwidth} % scratch, for one measurement
\newlength{\cvmaxcolwidth}

% A listing. The group is what keeps one section's measurements from leaking
% into the next.
\newenvironment{cvlisting}{%
  \par
  \setlength{\cvcolwidth}{0pt}%
  % A label past this takes its own line rather than squeezing the entry into
  % what is left, so one outlier cannot narrow the whole listing.
  \setlength{\cvmaxcolwidth}{0.32\linewidth}%
}{%
  \par
}

% Grow the column to fit one label. No output, and vertical mode throughout,
% so the measuring loop contributes nothing to the page.
\newcommand{\cvmeasure}[1]{%
  \settowidth{\cvlabelwidth}{#1}%
  \ifdim\cvlabelwidth>\cvcolwidth
    \ifdim\cvlabelwidth>\cvmaxcolwidth
      \setlength{\cvcolwidth}{\cvmaxcolwidth}%
    \else
      \setlength{\cvcolwidth}{\cvlabelwidth}%
    \fi
  \fi}

% The measured column, plus its gap. A listing whose labels are all empty
% measures 0pt and gets no gap either, so it sets flush left rather than
% carrying an empty column.
\newcommand{\cvsetcolumn}{%
  \setlength{\cvcoltotal}{\cvcolwidth}%
  \ifdim\cvcoltotal>0pt\addtolength{\cvcoltotal}{\cvlabelsep}\fi}

% One entry: #1 set in a column of width #2, #3 wrapping beside it with a
% hanging indent so continuation lines stay clear of the column.
%
% A label wider than its column would otherwise overlap the entry, since
% \makebox does not clip -- a real case, as a date such as
% `2001--2004, 2019--` is far wider than a plain year. So it is measured
% first, and an over-long one takes the whole line with the entry starting
% beneath it. \hangafter=1 already indents that second line to the column.
\newcommand{\cvcolumnentry}[3]{%
  \par\noindent
  \settowidth{\cvlabelwidth}{#1}%
  \ifdim\cvlabelwidth>#2\relax
    % Over-long label: it takes a line of its own, and the entry follows as a
    % paragraph indented to the column throughout (\hangafter=0 indents from
    % the first line, so no \hspace is needed and none may be added).
    #1\par\noindent
    \hangindent=#2\hangafter=0
    #3\par
  \else
    % Label fits: the entry sits beside it, and only its continuation lines
    % are indented clear of the column.
    \hangindent=#2\hangafter=1
    \makebox[#2][l]{#1}#3\par
  \fi}

% \cvsetcolumn is called per entry rather than once between the two loops:
% it is idempotent and costs nothing, and it saves the listings from having to
% emit a separator command between them.
\newcommand{\cventryline}[2]{\cvsetcolumn\cvcolumnentry{#1}{\cvcoltotal}{#2}}
\newcommand{\cvskillline}[2]{\cvsetcolumn\cvcolumnentry{\textbf{#1}}{\cvcoltotal}{#2}}

% Measuring counterparts, matching the above so that what is measured is what
% is later set: a skills label is bold, and bold is wider.
\newcommand{\cvmeasureentry}[1]{\cvmeasure{#1}}
\newcommand{\cvmeasureskill}[1]{\cvmeasure{\textbf{#1}}}

% The `why` elaboration under a detailed entry, indented to the entry column.
%
% beginpenalty forbids a page break between an entry and the first of its
% bullets, which otherwise happens readily and strands the entry alone at the
% foot of a page -- something upstream got for free by keeping an entry inside
% one table row. Breaks *between* bullets stay allowed, so a long elaboration
% can still split rather than forcing the whole entry to the next page.
\newenvironment{cvitems}{%
  \begin{itemize}[
    topsep=0.2ex, partopsep=0pt, itemsep=0.2ex, parsep=0pt,
    beginpenalty=10000,
    leftmargin=\dimexpr\cvcoltotal+1.2em\relax, labelsep=0.5em]%
}{%
  \end{itemize}%
}
